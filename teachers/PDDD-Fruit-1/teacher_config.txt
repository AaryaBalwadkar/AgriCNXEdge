from dataclasses import dataclass
from typing import List
import os


@dataclass
class DataConfig:
    data_root: str = ""
    use_existing_split: bool = False
    task_name: str = "pest"
    image_size: int = 224
    min_image_dim: int = 64
    min_pixel_std: float = 5.0
    phash_threshold: int = 8
    val_split: float = 0.15
    test_split: float = 0.15
    seed: int = 42
    batch_size: int = 32
    num_workers: int = 4
    pin_memory: bool = True


@dataclass
class ModelConfig:
    pddd_weights_path: str = ""
    pddd_num_classes: int = 120        # DO NOT CHANGE — original PDDD classes
    backbone_name: str = "convnext_base"
    feature_dim: int = 768          # DO NOT CHANGE — ConvNeXt-Small feature dim
    dropout_rate: float = 0.5


@dataclass
class TrainConfig:
    output_dir: str = ""
    task_name: str = "pest"
    num_classes: int = 4              # Updated per task below

    epochs: int = 50
    gradient_accumulation_steps: int = 1
    max_grad_norm: float = 1.0

    batch_size: int = 32
    num_workers: int = 4

    lr_backbone: float = 2e-6        # Fine-tuning LR — PDDD backbone
    lr_head: float = 5e-4            # Training LR — new classification head
    weight_decay: float = 1e-4

    eta_min: float = 1e-8

    focal_gamma: float = 2.0

    patience: int = 8
    min_delta: float = 1e-4

    quality_gate_f1: float = 0.70

    save_top_k: int = 1
    soft_label_temperature: float = 1.0
