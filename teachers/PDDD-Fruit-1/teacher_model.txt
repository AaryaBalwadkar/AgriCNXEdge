def remap_pddd_keys(old_sd: Dict) -> Dict:
    """
    PDDD was saved with timm <= 0.6.x naming. Current timm >= 0.9.x uses
    different names. This function converts all keys so the checkpoint
    loads correctly into the current timm ConvNeXt-Small.

    Mapping:
      downsample_layers.0.X.*     → stem.X.*
      downsample_layers.{i}.X.*  → stages.{i-1}.downsample.X.*   (i >= 1)
      stages.{i}.{j}.dwconv.*    → stages.{i}.blocks.{j}.conv_dw.*
      stages.{i}.{j}.pwconv1.*   → stages.{i}.blocks.{j}.mlp.fc1.*
      stages.{i}.{j}.pwconv2.*   → stages.{i}.blocks.{j}.mlp.fc2.*
      stages.{i}.{j}.norm.*      → stages.{i}.blocks.{j}.norm.*
      stages.{i}.{j}.gamma       → stages.{i}.blocks.{j}.gamma
      norm.weight / norm.bias    → head.norm.weight / head.norm.bias
      head.weight / head.bias    → head.fc.weight / head.fc.bias
    """
    new_sd = {}
    n_renamed = 0

    for old_key, value in old_sd.items():
        k = old_key

        # Rule 1: Stem  downsample_layers.0.X → stem.X
        k = re.sub(r'^downsample_layers\.0\.(\d+)(.*)', r'stem.\1\2', k)

        # Rule 2: Inter-stage downsamples  downsample_layers.{i}.X (i>=1)
        def _ds(m):
            return (f'stages.{int(m.group(1))-1}'
                    f'.downsample.{m.group(2)}{m.group(3)}')
        k = re.sub(r'^downsample_layers\.([1-9]\d*)\.(\d+)(.*)', _ds, k)

        # Rule 3: Stage blocks  stages.{i}.{j}.* → stages.{i}.blocks.{j}.*
        def _blk(m):
            suffix = m.group(3)
            suffix = re.sub(r'^dwconv\.',  'conv_dw.',  suffix)
            suffix = re.sub(r'^pwconv1\.', 'mlp.fc1.',  suffix)
            suffix = re.sub(r'^pwconv2\.', 'mlp.fc2.',  suffix)
            return f'stages.{m.group(1)}.blocks.{m.group(2)}.{suffix}'
        k = re.sub(r'^stages\.(\d+)\.(\d+)\.(.*)', _blk, k)

        # Rule 4: Final norm
        k = re.sub(r'^norm\.(weight|bias)$', r'head.norm.\1', k)

        # Rule 5: Head linear
        if k == 'head.weight': k = 'head.fc.weight'
        if k == 'head.bias':   k = 'head.fc.bias'

        if k != old_key:
            n_renamed += 1
        new_sd[k] = value

    total = len(old_sd)
    print(f"[INFO] Key remapping: {n_renamed}/{total} keys renamed, "
          f"{total - n_renamed} unchanged.")
    return new_sd


def filter_compatible_state_dict(
    model_state: Dict[str, torch.Tensor],
    checkpoint_state: Dict[str, torch.Tensor],
    skip_prefixes: List[str] = None,
) -> Tuple[Dict[str, torch.Tensor], List[Tuple[str, Tuple[int, ...], Tuple[int, ...]]], List[str]]:
    """Return checkpoint keys whose shapes match the model state dict."""
    skip_prefixes = skip_prefixes or []
    compatible = {}
    mismatched = []
    unexpected = []

    for k, v in checkpoint_state.items():
        if any(k.startswith(prefix) for prefix in skip_prefixes):
            continue
        if k not in model_state:
            unexpected.append(k)
            continue
        if model_state[k].shape == v.shape:
            compatible[k] = v
        else:
            mismatched.append((k, tuple(v.shape), tuple(model_state[k].shape)))

    return compatible, mismatched, unexpected

class FocalLoss(nn.Module):
    def __init__(self, gamma: float = 2.0, ignore_index: int = -1):
        super().__init__()
        self.gamma        = gamma
        self.ignore_index = ignore_index
        self.ce = nn.CrossEntropyLoss(reduction="none",
                                      ignore_index=ignore_index)

    def forward(self, logits: torch.Tensor, targets: torch.Tensor) -> torch.Tensor:
        ce   = self.ce(logits, targets)
        pt   = torch.exp(-ce)
        fl   = (1.0 - pt) ** self.gamma * ce
        mask = (targets != self.ignore_index).float()
        return (fl * mask).sum() / mask.sum().clamp(min=1.0)

def detect_convnext_variant(state_dict: Dict[str, torch.Tensor]) -> Tuple[str, int]:
    """Infer ConvNeXt variant and feature dim from checkpoint stem shape."""
    stem_key = None
    for key in state_dict.keys():
        if key.endswith("stem.0.weight") or key.endswith("stem.weight"):
            stem_key = key
            break

    if stem_key is None:
        print("[WARN] Could not detect stem weight. Defaulting to convnext_small.")
        return "convnext_small", 768

    out_channels = state_dict[stem_key].shape[0]
    if out_channels == 96:
        return "convnext_small", 768
    if out_channels == 128:
        return "convnext_base", 1024
    if out_channels == 192:
        return "convnext_large", 1536
    if out_channels == 48:
        return "convnext_nano", 384

    print(f"[WARN] Unknown stem output channels: {out_channels}. Defaulting to convnext_small.")
    return "convnext_small", 768


def detect_checkpoint_classes(state_dict: Dict[str, torch.Tensor]) -> int:
    """Return the number of classes encoded by the checkpoint head, if present."""
    for key, tensor in state_dict.items():
        if key.endswith("head.fc.weight") or key.endswith("head.weight"):
            return tensor.shape[0]
    return 40


class PDDDTeacher(nn.Module):
    def __init__(self, num_classes: int, pddd_weights_path: str, dropout_rate: float = 0.5):
        super().__init__()
        self.num_classes = num_classes

        if not os.path.exists(pddd_weights_path):
            raise FileNotFoundError(
                f"PDDD weights not found: {pddd_weights_path}\n"
                "Download: https://zenodo.org/records/7890438/files/"
                "convnext_small-model-99.pth?download=1"
            )

        print(f"[INFO] Inspecting PDDD checkpoint: {pddd_weights_path}")
        ckpt = torch.load(pddd_weights_path, map_location="cpu", weights_only=False)

        if isinstance(ckpt, dict):
            raw = (ckpt.get("model")
                   or ckpt.get("model_state_dict")
                   or ckpt.get("state_dict")
                   or ckpt)
        else:
            raw = ckpt

        raw = {k.replace("module.", ""): v for k, v in raw.items()}

        print("[INFO] Remapping PDDD checkpoint keys...")
        remapped = remap_pddd_keys(raw)

        detected_variant, feat_dim = detect_convnext_variant(remapped)
        checkpoint_classes = detect_checkpoint_classes(remapped)
        print(f"[AUTO-DETECT] Model variant: {detected_variant}, checkpoint classes: {checkpoint_classes}")

        self.model = timm.create_model(
            detected_variant,
            pretrained=False,
            num_classes=checkpoint_classes,
            drop_rate=dropout_rate,
        )

        model_state = self.model.state_dict()
        compatible, mismatched, unexpected = filter_compatible_state_dict(
            model_state,
            remapped,
            skip_prefixes=["head.fc.", "head.weight", "head.bias"],
        )

        if mismatched:
            print(f"[WARN] {len(mismatched)} checkpoint keys have incompatible shapes:")
            for key, ckpt_shape, model_shape in mismatched[:10]:
                print(f"  {key}: checkpoint={ckpt_shape} model={model_shape}")
            if len(mismatched) > 10:
                print(f"  ... and {len(mismatched)-10} more")

        if unexpected:
            print(f"[WARN] {len(unexpected)} checkpoint keys do not exist in the current model:")
            for key in unexpected[:10]:
                print(f"  {key}")
            if len(unexpected) > 10:
                print(f"  ... and {len(unexpected)-10} more")

        missing, unexpected2 = self.model.load_state_dict(compatible, strict=False)
        bb_missing = [k for k in missing if not k.startswith("head.fc")]
        bb_unexpected = [k for k in unexpected2 if not k.startswith("head.fc")]

        if bb_missing:
            print(f"[WARN] Missing backbone keys ({len(bb_missing)}): {bb_missing[:5]}")
        if bb_unexpected:
            print(f"[WARN] Unexpected backbone keys ({len(bb_unexpected)}): {bb_unexpected[:5]}")

        total_backbone = len([k for k in model_state if not k.startswith("head.fc")])
        print(f"[INFO] Loaded {len(compatible)}/{total_backbone} compatible backbone tensors.")

        if len(compatible) < total_backbone * 0.90:
            print("[ERROR] < 90% of backbone weights are compatible. "
                  "Check checkpoint variant and timm version.")

        in_features = self.model.head.fc.in_features
        print(f"[INFO] Replacing head: Linear({in_features}, {checkpoint_classes}) "
              f"→ Linear({in_features}, {num_classes})")
        self.model.head.fc = nn.Linear(in_features, num_classes)
        nn.init.normal_(self.model.head.fc.weight, std=0.01)
        nn.init.zeros_(self.model.head.fc.bias)

        print(f"[INFO] Model ready. Head: {num_classes} classes.")

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        return self.model(x)

    def get_parameter_groups(self, lr_backbone: float, lr_head: float, weight_decay: float):
        backbone_params = []
        head_params = []
        for name, param in self.named_parameters():
            if "head.fc" in name:
                head_params.append(param)
            else:
                backbone_params.append(param)

        return [
            {"params": backbone_params, "lr": lr_backbone, "weight_decay": weight_decay, "name": "backbone"},
            {"params": head_params,     "lr": lr_head,     "weight_decay": 0.0,            "name": "head"},
        ]
